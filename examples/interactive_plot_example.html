import numpy as np
import plotly.graph_objects as go

# Base x values
x = np.linspace(0, 2 * np.pi, 400)

# Values the slider will step through
a_values = np.linspace(0.5, 5, 20)

# Create frames
frames = []
for a in a_values:
    y = np.sin(a * x)
    frames.append(
        go.Frame(
            data=[go.Scatter(x=x, y=y)],
            name=f"{a:.2f}"
        )
    )

# Initial figure
fig = go.Figure(
    data=[go.Scatter(x=x, y=np.sin(a_values[0] * x))],
    layout=go.Layout(
        title="y = sin(a · x)",
        xaxis=dict(title="x"),
        yaxis=dict(title="y"),
        sliders=[{
            "active": 0,
            "currentvalue": {"prefix": "a = "},
            "steps": [
                {
                    "label": f"{a:.2f}",
                    "method": "animate",
                    "args": [
                        [f"{a:.2f}"],
                        {"mode": "immediate", "frame": {"duration": 0}}
                    ],
                }
                for a in a_values
            ],
        }],
    ),
    frames=frames
)

HTML(fig.to_html(include_plotlyjs="cdn"))
